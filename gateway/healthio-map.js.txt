const a = msg.payload;
const unknown = () => ({topic:'controller-health',payload:[{name:'pac_io_communication_ok',value:-1}]});
if (msg.error || !Array.isArray(a) || a.length < 14 ||
    !a.slice(0,14).every(Number.isInteger) || a[0] !== 1 || a[1] <= 0 || a[1] % 2 !== 0) {
  context.set('previousSequence', null);
  return unknown();
}
const previous = context.get('previousSequence');
context.set('previousSequence', a[1]);
if (previous == null) return unknown();
// Do not refresh the timestamp on a frozen health chart. Website expires it.
if (previous === a[1]) return null;
let value = -1;
if (a[4] === 0 && (a[2] === 2 || a[2] === 3)) value = 0;
if (a[4] === 0 && a[2] === 1 && a[3] === 1 &&
    a[5] === 0 && a[6] === 0 && a[7] === 0 && a[13] === 1) {
  if (a[12] === 1) value = 1;
  else if (a[12] === 2) value = 0;
}
return {topic:'controller-health',payload:[{name:'pac_io_communication_ok',value}]};
